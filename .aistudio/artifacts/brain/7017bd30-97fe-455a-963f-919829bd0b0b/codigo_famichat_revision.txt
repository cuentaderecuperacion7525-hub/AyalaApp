import React, { useState } from 'react';
import { Contact } from '../types';
import { ShieldCheck, Lock, Check, HelpCircle, ArrowLeft, KeyRound, AlertCircle } from 'lucide-react';

interface WelcomeGatewayProps {
  contacts: Contact[];
  onSelectUser: (user: Contact) => void;
}

export const WelcomeGateway: React.FC<WelcomeGatewayProps> = ({ contacts, onSelectUser }) => {
  const [selectedCandidate, setSelectedCandidate] = useState<Contact | null>(null);
  const [pinCode, setPinCode] = useState<string>('');
  const [pinError, setPinError] = useState<string>('');
  const [isSuccess, setIsSuccess] = useState<boolean>(false);

  const getExpectedPin = () => {
    return localStorage.getItem('famichat_admin_pin') || '1122';
  };

  const handleSelect = (user: Contact) => {
    if (user.role === 'admin') {
      setSelectedCandidate(user);
      setPinCode('');
      setPinError('');
      setIsSuccess(false);
    } else {
      // Normal family members enter directly with absolute privacy active
      onSelectUser(user);
    }
  };

  const handleKeypadPress = (num: string) => {
    setPinError('');
    if (pinCode.length >= 4) return;
    const nextPin = pinCode + num;
    setPinCode(nextPin);

    const expected = getExpectedPin();
    if (nextPin === expected) {
      setIsSuccess(true);
      setTimeout(() => {
        if (selectedCandidate) {
          onSelectUser(selectedCandidate);
        }
      }, 700);
    } else if (nextPin.length === expected.length) {
      setTimeout(() => {
        setPinError('Clave de administrador incorrecta. Acceso denegado.');
        setPinCode('');
      }, 150);
    }
  };

  const handleBackspace = () => {
    setPinCode(prev => prev.slice(0, -1));
    setPinError('');
  };

  const KEYPAD_NUMBERS = ['1', '2', '3', '4', '5', '6', '7', '8', '9'];

  if (selectedCandidate) {
    return (
      <div
        id="welcome-identity-gateway"
        className="fixed inset-0 h-[100dvh] w-full bg-slate-900 bg-gradient-to-tr from-slate-950 via-teal-950 to-emerald-950 text-white z-[99999] flex flex-col overflow-hidden select-none"

          paddingTop: 'calc(max(env(safe-area-inset-top, 0px), var(--sat, 0px)) + 12px)',
          paddingBottom: 'calc(max(env(safe-area-inset-bottom, 0px), var(--sab, 0px)) + 16px)',
        }}
      >
        <div className="w-full max-w-[430px] h-full mx-auto px-4 flex flex-col justify-between">
          {/* BLOQUE 1 (ARRIBA): Logo/Título y puntos del PIN */}
          <div className="w-full shrink-0 flex flex-col items-center">
            <div className="w-full flex items-center justify-between mb-2">
              <button
                type="button"
                onClick={() => setSelectedCandidate(null)}
                className="text-slate-300 active:bg-white/15 flex items-center gap-1.5 text-xs font-bold bg-white/10 px-3 py-1.5 rounded-[12px] transition-colors duration-75 cursor-pointer"
              >
                <ArrowLeft className="w-4 h-4" />
                <span>Volver</span>

                {selectedCandidate.avatarEmoji}

import { Capacitor, CapacitorHttp } from '@capacitor/core';
import { App as CapApp } from '@capacitor/app';

export const GITHUB_OWNER = "cuentaderecuperacion7525-hub";
export const GITHUB_REPO = "AyalaApp";

const WEB_FALLBACK_VERSION = "1.0.0";

export interface ReleaseInfo {
  version: string;      // Siempre con prefijo "v", ej. "v1.0.15"
  cleanVersion: string; // Sin prefijo "v", ej. "1.0.15"
  body: string;
  downloadUrl: string;
  releaseUrl: string;
}

export async function getInstalledAppVersion(): Promise<string> {
  try {
    if (Capacitor.isNativePlatform()) {
      const info = await CapApp.getInfo();
      if (info && info.version) {
        return info.version.trim().replace(/^v/i, '');
      }
    }
  } catch (e) {
    console.warn('No se pudo leer App.getInfo(), usando versión de respaldo:', e);
  }
  return WEB_FALLBACK_VERSION;
}

export function compareVersions(remoteTag: string, currentVersion: string): number {
  const sanitize = (v: string) => v.trim().replace(/^v/i, '');
  const p1 = sanitize(remoteTag).split('.').map(n => parseInt(n, 10) || 0);
  const p2 = sanitize(currentVersion).split('.').map(n => parseInt(n, 10) || 0);
  const maxLen = Math.max(p1.length, p2.length);

  for (let i = 0; i < maxLen; i++) {
    const num1 = p1[i] ?? 0;
    const num2 = p2[i] ?? 0;
    if (num1 > num2) return 1;
    if (num1 < num2) return -1;
  }
  return 0;
}

export async function fetchLatestRelease(installedVersionOverride?: string): Promise<{
  success: boolean;
  hasUpdate: boolean;
  installedVersion: string;
  release?: ReleaseInfo;
  errorMessage?: string;
}> {
  const installedVersion = installedVersionOverride || (await getInstalledAppVersion());

  if (typeof navigator !== 'undefined' && !navigator.onLine) {
    return {
      success: false,
      hasUpdate: false,
      installedVersion,
      errorMessage: 'Sin internet'
    };
  }

  try {
    const url = `https://api.github.com/repos/${GITHUB_OWNER}/${GITHUB_REPO}/releases/latest`;
    const response = await CapacitorHttp.get({
      url,
      headers: {
        'Accept': 'application/vnd.github.v3+json',
        'User-Agent': 'FamiChat-App'
      }
    });

    if (response.status === 404) {
      return {
        success: false,
        hasUpdate: false,
        installedVersion,
        errorMessage: 'No se encontró ninguna release'
      };
    }

    if (response.status < 200 || response.status >= 300) {
      return {
        success: false,
        hasUpdate: false,
        installedVersion,
        errorMessage: `Error al consultar GitHub (código ${response.status})`
      };
    }

    const data = typeof response.data === 'string' ? JSON.parse(response.data) : response.data;
    const rawTag = (data?.tag_name || '').trim();

    if (!rawTag) {
      return {
        success: false,
        hasUpdate: false,
        installedVersion,
        errorMessage: 'No se encontró ninguna release'
      };
    }

    const cleanTag = rawTag.replace(/^v/i, '');
    const displayTag = `v${cleanTag}`;
    const isNewer = compareVersions(cleanTag, installedVersion) > 0;

    let apkDownloadUrl = '';
    if (Array.isArray(data.assets)) {
      const apkAsset = data.assets.find(
        (asset: any) => typeof asset?.name === 'string' && asset.name.toLowerCase().endsWith('.apk')
      );
      if (apkAsset?.browser_download_url) {
        apkDownloadUrl = apkAsset.browser_download_url;
      }
    }

    const fallbackReleaseUrl = data.html_url || `https://github.com/${GITHUB_OWNER}/${GITHUB_REPO}/releases/latest`;

    return {
      success: true,
      hasUpdate: isNewer,
      installedVersion,
      release: {
        version: displayTag,
        cleanVersion: cleanTag,
        body: data.body || 'Nuevas mejoras y optimizaciones para la familia.',
        downloadUrl: apkDownloadUrl || fallbackReleaseUrl,
        releaseUrl: fallbackReleaseUrl
      }
    };
  } catch (err) {
    const isOffline = typeof navigator !== 'undefined' && !navigator.onLine;
    return {
      success: false,
      hasUpdate: false,
      installedVersion,
      errorMessage: isOffline ? 'Sin internet' : 'No se pudo comprobar. Revisa tu conexión.'
    };
  }
}

    localStorage.setItem('famichat_auto_check_updates', val ? 'true' : 'false');
  };

  // Check for updates once on startup in background with a 3-second delay after app loads,
  // only if there is internet. If it fails, retry once after 10 seconds.
  const hasStartedAutoUpdateCheckRef = useRef<boolean>(false);
  useEffect(() => {
    if (loading || hasStartedAutoUpdateCheckRef.current) return;
    hasStartedAutoUpdateCheckRef.current = true;

    let isCancelled = false;
    let retryTimer: ReturnType<typeof setTimeout> | null = null;

    const performCheck = async (isRetry: boolean) => {
      const isAutoEnabled = localStorage.getItem('famichat_auto_check_updates') !== 'false';
      if (!isAutoEnabled) return;

      if (typeof navigator !== 'undefined' && !navigator.onLine) {
        if (!isRetry) {
          retryTimer = setTimeout(() => {
            if (!isCancelled) performCheck(true);
          }, 10000);
        }
        return;
      }

      const result = await fetchLatestRelease();
      if (isCancelled) return;

      if (result.installedVersion) {
        setInstalledVersion(result.installedVersion);
      }

      if (!result.success) {
        if (!isRetry) {
          retryTimer = setTimeout(() => {
            if (!isCancelled) performCheck(true);
          }, 10000);
        }
        return;
      }

      if (result.hasUpdate && result.release) {
        setHasUpdate(true);
        setLatestRelease(result.release);

        const lastShownVersion = localStorage.getItem('famichat_dismissed_update_version');

        // Show dialog once per version ONLY if user is logged in and on the chat list (never inside a chat)
        if (
          activeUserRef.current &&
          mobileViewRef.current === 'list' &&
          lastShownVersion !== result.release.version
        ) {
          localStorage.setItem('famichat_dismissed_update_version', result.release.version);
          window.history.pushState({ view: 'update_modal' }, '');
          setShowUpdateModal(true);
        }
      }
    };

    const initialTimer = setTimeout(() => {
      if (!isCancelled) performCheck(false);
    }, 3000);

    return () => {
      isCancelled = true;
      clearTimeout(initialTimer);
      if (retryTimer) clearTimeout(retryTimer);
    };
  }, [loading]);

  const handleManualCheckUpdates = async () => {
    setCheckingUpdate(true);
    setUpdateCheckFeedback(null);

    const result = await fetchLatestRelease();
    setCheckingUpdate(false);

    const currentVer = result.installedVersion || installedVersion;
    if (result.installedVersion) {
      setInstalledVersion(result.installedVersion);
    }

    if (result.success) {
      if (result.hasUpdate && result.release) {
        setHasUpdate(true);
        setLatestRelease(result.release);
        setUpdateCheckFeedback({
          type: 'available',
          message: `Hay una nueva: v${result.release.version}`,
        });
        window.history.pushState({ view: 'update_modal' }, '');
        setShowUpdateModal(true);
      } else {
        setHasUpdate(false);
        setUpdateCheckFeedback({
          type: 'uptodate',
          message: `Estás al día (versión ${currentVer})`,
        });
      }
    } else {
      setUpdateCheckFeedback({
        type: 'error',
        message: result.errorMessage || 'Sin internet',
      });
    }
  };
