import React, { useState } from 'react';
import { Contact } from '../types';
import { ShieldCheck, Lock, Check, HelpCircle, ArrowLeft, KeyRound, AlertCircle, Delete } from 'lucide-react';

interface WelcomeGatewayProps {
  contacts: Contact[];
  onSelectUser: (user: Contact) => void;
}

// Letras decorativas bajo cada número (solo diseño, el PIN sigue siendo numérico)
const KEY_LETTERS: Record<string, string> = {
  '1': '', '2': 'ABC', '3': 'DEF', '4': 'GHI', '5': 'JKL',
  '6': 'MNO', '7': 'PQRS', '8': 'TUV', '9': 'WXYZ', '0': '',
};

const KEY_HEIGHT = 'clamp(60px, 9.5dvh, 84px)';

export const WelcomeGateway: React.FC<WelcomeGatewayProps> = ({ contacts, onSelectUser }) => {
  const [selectedCandidate, setSelectedCandidate] = useState<Contact | null>(null);
  const [pinCode, setPinCode] = useState<string>('');
  const [pinError, setPinError] = useState<string>('');
  const [isSuccess, setIsSuccess] = useState<boolean>(false);
  const [shake, setShake] = useState<boolean>(false); // solo animación visual

  const getExpectedPin = () => {
    return localStorage.getItem('famichat_admin_pin') || '1122';
  };

  const handleSelect = (user: Contact) => {
    if (user.role === 'admin') {
      setSelectedCandidate(user);
      setPinCode('');
      setPinError('');
      setIsSuccess(false);
    } else {
      // Normal family members enter directly with absolute privacy active
      onSelectUser(user);
    }
  };

  // Retroalimentación visual/táctil al fallar (no afecta la lógica)
  const triggerErrorFeedback = () => {
    setShake(true);
    setTimeout(() => setShake(false), 350);
    try {
      if (typeof navigator !== 'undefined' && 'vibrate' in navigator) {
        navigator.vibrate(120);
      }
    } catch {
      /* si el dispositivo no vibra, se ignora */
    }
  };

  const handleKeypadPress = (num: string) => {
    setPinError('');
    const expected = getExpectedPin();
    if (pinCode.length >= expected.length) return;
    const nextPin = pinCode + num;
    setPinCode(nextPin);

    if (nextPin === expected) {
      setIsSuccess(true);
      setTimeout(() => {
        if (selectedCandidate) {
          onSelectUser(selectedCandidate);
        }
      }, 700);
    } else if (nextPin.length === expected.length) {
      setTimeout(() => {
        setPinError('Clave de administrador incorrecta. Acceso denegado.');
        setPinCode('');
        triggerErrorFeedback();
      }, 150);
    }
  };

  const handleBackspace = () => {
    setPinCode(prev => prev.slice(0, -1));
    setPinError('');
  };

  const KEYPAD_NUMBERS = ['1', '2', '3', '4', '5', '6', '7', '8', '9'];

  // Tecla numérica: número arriba + letras pequeñas debajo (línea siempre reservada)
  const renderNumberKey = (num: string) => (
    <button
      key={num}
      type="button"
      disabled={isSuccess}
      onClick={() => handleKeypadPress(num)}
      aria-label={`Número ${num}`}
      style={{ height: KEY_HEIGHT }}
      className="w-full rounded-[18px] flex flex-col items-center justify-center bg-slate-900/70 border border-slate-700/60 text-white transition-all duration-100 active:scale-95 active:bg-emerald-400/15 active:border-emerald-400/50 focus:outline-none focus-visible:ring-2 focus-visible:ring-emerald-400 disabled:opacity-60 cursor-pointer select-none"
    >
      <span className="text-[30px] font-semibold leading-none tabular-nums">{num}</span>
      <span className="mt-1.5 h-[11px] text-[11px] leading-none tracking-[0.2em] text-slate-400">
        {KEY_LETTERS[num]}
      </span>
    </button>
  );

  if (selectedCandidate) {
    const expectedLength = getExpectedPin().length;

    return (
      <div
        id="welcome-identity-gateway"
        className="fixed inset-0 h-[100dvh] w-full bg-slate-900 bg-gradient-to-tr from-slate-950 via-teal-950 to-emerald-950 text-white z-[99999] flex flex-col overflow-hidden select-none"
        style={{
          paddingTop: 'calc(max(env(safe-area-inset-top, 0px), var(--sat, 0px)) + 12px)',
          paddingBottom: 'calc(max(env(safe-area-inset-bottom, 0px), var(--sab, 0px)) + 16px)',
        }}
      >
        <style>{`
          @keyframes fc-shake {
            0%, 100% { transform: translateX(0); }
            20% { transform: translateX(-8px); }
            40% { transform: translateX(8px); }
            60% { transform: translateX(-6px); }
            80% { transform: translateX(6px); }
          }
          .fc-shake { animation: fc-shake 300ms ease-in-out; }
          @media (prefers-reduced-motion: reduce) {
            .fc-shake { animation: none; }
          }
        `}</style>

        {/* Tarjeta contenedora con borde, como la referencia */}
        <div className="w-full max-w-[430px] flex-1 min-h-0 mx-auto px-3">
          <div className="w-full h-full rounded-[28px] border border-slate-700/60 bg-slate-950/70 backdrop-blur px-4 pt-4 pb-2 flex flex-col overflow-y-auto">

            {/* BLOQUE 1 (ARRIBA): Título, avatar y puntos del PIN */}
            <div className="w-full shrink-0 flex flex-col items-center">
              <span className="text-xs font-black tracking-[0.2em] uppercase text-emerald-300/90 font-sans">
                FamiChat
              </span>

              <div className="flex flex-col items-center text-center mt-3">
                <div
                  className={`flex items-center justify-center rounded-[20px] border border-emerald-400/40 shadow-md text-4xl ${selectedCandidate.avatarColor}`}
                  style={{ width: 'clamp(52px, 8dvh, 72px)', height: 'clamp(52px, 8dvh, 72px)' }}
                >
                  {selectedCandidate.avatarEmoji}
                </div>

                <h3 className="mt-3 font-bold text-white text-[22px] leading-tight">
                  {selectedCandidate.name}
                </h3>

                <p className="mt-1 text-[11px] text-amber-400 font-semibold tracking-[0.18em] uppercase flex items-center gap-1.5 justify-center">
                  <Lock className="w-3.5 h-3.5" />
                  <span>Acceso Protegido de Administrador</span>
                </p>

                <p className="mt-2 text-sm text-slate-300">Ingresa tu PIN</p>
              </div>

              {/* Puntos del PIN */}
              <div className={`flex justify-center gap-5 mt-4 ${shake ? 'fc-shake' : ''}`}>
                {Array.from({ length: expectedLength }).map((_, idx) => (
                  <div
                    key={idx}
                    className={`w-4 h-4 rounded-full border-2 transition-colors duration-100 ${
                      idx < pinCode.length
                        ? 'bg-emerald-400 border-emerald-400 shadow-[0_0_10px_#34d399]'
                        : 'border-slate-600 bg-slate-800'
                    }`}
                  />
                ))}
              </div>

              {/* Mensajes (altura reservada para que el teclado no salte) */}
              <div className="w-full min-h-[40px] mt-3 flex items-center justify-center" aria-live="polite">
                {pinError && (
                  <div className="flex items-center gap-1.5 text-xs text-rose-300 font-semibold bg-rose-950/60 border border-rose-800 rounded-[12px] px-3 py-2 justify-center text-center">
                    <AlertCircle className="w-4 h-4 shrink-0" />
                    <span>{pinError}</span>
                  </div>
                )}
                {isSuccess && (
                  <div className="flex items-center gap-1.5 text-xs text-emerald-300 font-bold bg-emerald-950/60 border border-emerald-800 rounded-[12px] px-3 py-2 justify-center text-center">
                    <ShieldCheck className="w-4 h-4 text-emerald-400 shrink-0" />
                    <span>Llave Correcta. Iniciando...</span>
                  </div>
                )}
              </div>
            </div>

            {/* BLOQUE 2 (MEDIO): Espacio flexible */}
            <div className="flex-1 min-h-2" />

            {/* BLOQUE 3 (ABAJO): Teclado 3x4 */}
            <div className="w-full shrink-0">
              <div
                className="grid w-full"
                style={{ gridTemplateColumns: 'repeat(3, 1fr)', gap: '12px' }}
              >
                {KEYPAD_NUMBERS.map(renderNumberKey)}

                {/* Fila 4: [vacío]  0  [borrar] */}
                <div style={{ height: KEY_HEIGHT }} aria-hidden="true" />
                {renderNumberKey('0')}
                <button
                  type="button"
                  disabled={isSuccess}
                  onClick={handleBackspace}
                  aria-label="Borrar"
                  title="Borrar"
                  style={{ height: KEY_HEIGHT }}
                  className="w-full rounded-[18px] flex items-center justify-center bg-slate-900/40 border border-slate-700/40 text-slate-200 transition-all duration-100 active:scale-95 active:bg-emerald-400/15 active:border-emerald-400/50 focus:outline-none focus-visible:ring-2 focus-visible:ring-emerald-400 disabled:opacity-60 cursor-pointer select-none"
                >
                  <Delete className="w-7 h-7" />
                </button>
              </div>

              {/* Línea divisoria y botón Volver como enlace de texto */}
              <div className="mt-3 border-t border-slate-700/50">
                <button
                  type="button"
                  onClick={() => setSelectedCandidate(null)}
                  className="w-full h-12 flex items-center justify-center gap-2 text-sm font-medium text-slate-300 active:text-white transition-colors duration-100 cursor-pointer focus:outline-none focus-visible:ring-2 focus-visible:ring-emerald-400 rounded-[12px]"
                >
                  <ArrowLeft className="w-4 h-4" />
                  <span>Volver</span>
                </button>
              </div>
            </div>
          </div>
        </div>
      </div>
    );
  }

  // ─── PANTALLA DE SELECCIÓN DE FAMILIAR ───
  // NO MODIFICADO: aquí va tu `return ( ... )` original, tal como lo tienes.
  return ( ... );
};


import React, { useState } from 'react';
import { ReleaseInfo } from '../config/version';

interface UpdateModalProps {
  isOpen: boolean;
  release: ReleaseInfo | null;
  onClose: () => void;
  onUpdateNow: () => void;
}

export const UpdateModal: React.FC<UpdateModalProps> = ({
  isOpen,
  release,
  onClose,
  onUpdateNow
}) => {
  // Solo interfaz: muestra u oculta la ayuda del conflicto de instalación
  const [showInstallHelp, setShowInstallHelp] = useState(false);

  if (!isOpen || !release) return null;

  const formatBodyLines = (text: string) => {
    if (!text) return ['Nuevas mejoras de rendimiento y estabilidad para la familia.'];
    const lines = text
      .split('\n')
      .map(l => l.trim())
      .filter(Boolean);
    return lines.slice(0, 5);
  };

  const bodyLines = formatBodyLines(release.body);
  const versionLabel = release.version.startsWith('v') ? release.version : `v${release.version}`;

  return (
    <div
      id="update-modal-overlay"
      className="fixed inset-0 bg-slate-950/70 backdrop-blur-sm z-[10000] flex items-end justify-center select-none"
      style={{ animation: 'fc-fade 200ms ease-out' }}

      <style>{`
        @keyframes fc-fade { from { opacity: 0; } to { opacity: 1; } }
        @keyframes fc-slide-up { from { transform: translateY(100%); } to { transform: translateY(0); } }
        @media (prefers-reduced-motion: reduce) {
          #update-modal-overlay, #update-modal-card { animation: none !important; }
        }
      `}</style>

      <div
        id="update-modal-card"
        role="dialog"
        aria-modal="true"
        aria-labelledby="update-modal-title"
        className="w-full max-w-[430px] bg-white dark:bg-slate-900 text-slate-800 dark:text-slate-100 rounded-t-[24px] shadow-2xl border-t border-slate-200 dark:border-slate-800 overflow-hidden"

          animation: 'fc-slide-up 250ms cubic-bezier(0.2, 0.8, 0.2, 1)',

      >
        {/* Barra de arrastre */}
        <div className="pt-2.5 pb-1 flex justify-center">
          <div className="w-10 h-1 rounded-full bg-slate-300 dark:bg-slate-700" />
        </div>

        {/* Encabezado */}
        <div className="px-5 pt-2 pb-3 flex items-center gap-3">
          <div className="w-11 h-11 rounded-[12px] bg-gradient-to-br from-emerald-600 to-teal-600 flex items-center justify-center shrink-0 shadow-md shadow-emerald-700/20">
            <DownloadCloud className="w-5 h-5 text-white" />
          </div>

          </div>
          <button
            type="button"
            onClick={onClose}
            aria-label="Cerrar"
            title="Cerrar"
            className="w-11 h-11 -mr-2 rounded-[12px] flex items-center justify-center text-slate-500 dark:text-slate-400 active:bg-slate-100 dark:active:bg-slate-800 transition-colors cursor-pointer shrink-0"
          >
            <X className="w-5 h-5" />
          </button>
        </div>

        <div className="px-5 space-y-3">
          {/* Notas de la versión */}
          <div className="bg-emerald-50/60 dark:bg-slate-800/60 border border-emerald-100 dark:border-slate-700/60 rounded-[16px] p-4 space-y-2">
            <span className="text-xs font-bold uppercase tracking-wider text-emerald-700 dark:text-emerald-400 block">
              Notas de la versión
            </span>
            <div className="text-sm text-slate-600 dark:text-slate-300 space-y-1.5 leading-relaxed">
              {bodyLines.map((line, idx) => (
                <p key={idx} className="flex items-start gap-2">
                  <span className="text-emerald-500 font-bold shrink-0">•</span>
                  <span
                    style={{
                      display: '-webkit-box',
                      WebkitLineClamp: 2,
                      WebkitBoxOrient: 'vertical',
                      overflow: 'hidden',
                    }}
                  >
                    {line.replace(/^[-*#]\s*/, '')}
                  </span>
                </p>
              ))}
            </div>
          </div>

          {/* Indicación de instalación + ayuda plegable */}
          <div className="bg-teal-50 dark:bg-teal-950/40 px-4 py-3 rounded-[16px] border border-teal-200/50 dark:border-teal-800/40">
            <div className="flex items-center gap-2 text-sm font-semibold text-teal-700 dark:text-teal-300">

              <span>Toca Instalar al terminar de descargar.</span>
            </div>

            <button
              type="button"
              onClick={() => setShowInstallHelp(v => !v)}
              aria-expanded={showInstallHelp}

              <span>Actualizar ahora</span>
            </button>
          </div>
        </div>
      </div>
    </div>
  );
